\section*{Exercise 9}

Given $\mu I \preceq \nabla^2 f \preceq L I$, $0 < \mu \le L$, and
$T_\gamma(x) := x - \gamma \nabla f(x)$.

\subsection*{1)}
\[
  T_\gamma(x) - T_\gamma(y) = \int_0^1 \nabla T_\gamma\bigl(y + t(x-y)\bigr)(x-y)\,dt
  \;\Rightarrow
\]
\begin{align*}
  \norm{T_\gamma(x) - T_\gamma(y)}
  &= \norm{\int_0^1 \nabla T_\gamma\bigl(y + t(x-y)\bigr)(x-y)\,dt}
  \le \int_0^1 \norm{\nabla T_\gamma\bigl(y + t(x-y)\bigr)}\norm{x-y}\,dt \\
  &\le \int_0^1 \sup_{z \in \R^d}\norm{\nabla T_\gamma(z)}\,dt \cdot \norm{x-y}
  = \sup_{z \in \R^d}\norm{I - \gamma \nabla^2 f(z)}\,\norm{x-y}.
\end{align*}
Since $\mu I \preceq \nabla^2 f \preceq L I$, the eigenvalues $\lambda_i$ of the
symmetric matrix $\nabla^2 f(z)$ lie in $[\mu, L]$, so
\[
  \norm{I - \gamma \nabla^2 f(z)} = \max_i |1 - \gamma\lambda_i|
  \le \max\{|1-\gamma L|, |1-\mu\gamma|\}
\]
($|1-\gamma\lambda|$ is convex in $\lambda$, so its maximum on $[\mu, L]$ is at an endpoint)
$\Rightarrow$
\[
  \norm{T_\gamma(x) - T_\gamma(y)} \le \norm{x-y}\cdot\max\{|1-\gamma L|, |1-\mu\gamma|\}.
\]

\subsection*{2)}
If $\gamma = \frac{2}{\mu+L}$ $\Rightarrow$
$|1-\gamma L| = |1-\mu\gamma| = \frac{L-\mu}{L+\mu}$.

If $\gamma > \frac{2}{\mu+L}$ $\Rightarrow$ $L\gamma > \frac{2L}{\mu+L}$ $\Rightarrow$
$1 - L\gamma < 1 - \frac{2L}{\mu+L} = \frac{\mu-L}{\mu+L} \le 0$ $\Rightarrow$
$|1-L\gamma| > \frac{L-\mu}{L+\mu}$ $\Rightarrow$
$\max\{|1-\gamma L|, |1-\mu\gamma|\} > \frac{L-\mu}{L+\mu}$.

If $\gamma < \frac{2}{\mu+L}$ $\Rightarrow$ $\mu\gamma < \frac{2\mu}{\mu+L}$ $\Rightarrow$
$1 - \mu\gamma > 1 - \frac{2\mu}{\mu+L} = \frac{L-\mu}{L+\mu} \ge 0$ $\Rightarrow$
$|1-\mu\gamma| > \frac{L-\mu}{L+\mu}$ $\Rightarrow$
$\max\{|1-\gamma L|, |1-\mu\gamma|\} > \frac{L-\mu}{L+\mu}$.

$\Rightarrow$ $\gamma = \frac{2}{\mu+L}$ is optimal.

\subsection*{3)}
Since $\nabla f(\xs) = 0$, we have $T_\gamma(\xs) = \xs$, so with $\gamma = \frac{2}{\mu+L}$
\[
  \norm{x^{k+1} - \xs} = \norm{T_\gamma(x^k) - T_\gamma(\xs)}
  \le \frac{L-\mu}{L+\mu}\norm{x^k - \xs}
  \;\Rightarrow\;
  \norm{x^k - \xs} \le \norm{x^0 - \xs}\cdot\left(\frac{L-\mu}{L+\mu}\right)^k .
\]
In Theorem 15, $\norm{x^k - \xs}^2 \le (1-\gamma\mu)^k\norm{x^0 - \xs}^2$, where
$\gamma \le \frac{1}{L}$; the best choice is $\gamma = \frac{1}{L}$
\[
  \Rightarrow\; \norm{x^k - \xs}^2 \le \left(1 - \frac{\mu}{L}\right)^k\norm{x^0 - \xs}^2
  = \left(\frac{L-\mu}{L}\right)^k\norm{x^0 - \xs}^2 .
\]
Squaring our bound, $\norm{x^k - \xs}^2 \le \left(\frac{L-\mu}{L+\mu}\right)^{2k}\norm{x^0 - \xs}^2$, and
\[
  \frac{L-\mu}{L} : \left(\frac{L-\mu}{L+\mu}\right)^2
  = \frac{(L+\mu)^2}{L(L-\mu)} > 1
  \quad (\mu < L),
\]
so the rate with $\gamma = \frac{2}{\mu+L}$ is better, by the factor
$\frac{(L+\mu)^2}{L(L-\mu)}$ per iteration.

\section*{Exercise 10}

If $f$ is $L$-smooth and $\mu$-strongly convex, then by (18) and Exercise 8
\[
  \frac{1}{L}\norm{\nabla f(x^k) - \nabla f(\xs)}^2
  \le 2D_f(x^k, \xs)
  \le \frac{1}{\mu}\norm{\nabla f(x^k) - \nabla f(\xs)}^2 .
\]
Dividing by $\norm{\nabla f(x^k) - \nabla f(\xs)}^2$, (i) and (ii) are true:
$\frac{1}{L} \le \gamma_k \le \frac{1}{\mu}$.

For (iii): by (i), $1 - \gamma_k L \le 0$, so the last term of Lemma 14 cannot be
dropped. Instead we stop the proof of Lemma 14 before using (25), with
$r^k = x^k - \xs$:
\[
  \norm{r^{k+1}}^2 \le (1 - \gamma_k\mu)\norm{r^k}^2 - 2\gamma_k D_f(x^k, \xs)
  + \gamma_k^2\norm{\nabla f(x^k) - \nabla f(\xs)}^2 .
\]
By the definition of $\gamma_k$,
$\gamma_k^2\norm{\nabla f(x^k) - \nabla f(\xs)}^2 = 2\gamma_k D_f(x^k, \xs)$
$\Rightarrow$
\[
  \norm{x^{k+1} - \xs}^2 \le (1 - \gamma_k\mu)\norm{x^k - \xs}^2 .
\]

\section*{Exercise 11}

$f : \R^d \to \R$ is differentiable. Then for all $x, y, z \in \R^d$
\[
  D_f(x,z) = D_f(x,y) + \ip{x-y}{\nabla f(y) - \nabla f(z)} + D_f(y,z). \tag{a}
\]
This is the same as
\begin{align*}
  f(x) - f(z) - \ip{\nabla f(z)}{x-z}
  &= f(x) - f(y) - \ip{\nabla f(y)}{x-y} \\
  &\quad + \ip{x-y}{\nabla f(y) - \nabla f(z)} + f(y) - f(z) - \ip{\nabla f(z)}{y-z}.
\end{align*}
Cancel the common terms on both sides ($f(\cdot)$):
\[
  -\ip{\nabla f(z)}{x-z}
  = -\ip{\nabla f(y)}{x-y} - \ip{\nabla f(z)}{y-z} + \ip{x-y}{\nabla f(y) - \nabla f(z)}.
\]
Also $\ip{x-y}{\nabla f(y) - \nabla f(z)} = \ip{\nabla f(y)}{x-y} - \ip{\nabla f(z)}{x-y}$,
so this is the same as
\[
  -\ip{\nabla f(z)}{x-z} = -\ip{\nabla f(z)}{y-z} - \ip{\nabla f(z)}{x-y}
\]
$\iff$
\[
  \ip{-\nabla f(z)}{x - z + z - y + y - x} = \ip{-\nabla f(z)}{0} = 0,
\]
which is true.

\subsection*{(i)}
If $x = z$, then $D_f(x,z) = f(x) - f(x) - \ip{\nabla f(x)}{x-x} = 0$. As a result we get
\[
  D_f(x,y) + D_f(y,x) = -\ip{x-y}{\nabla f(y) - \nabla f(x)} = \ip{\nabla f(x) - \nabla f(y)}{x-y}.
\]

\subsection*{(ii)}
If $f(x) = \frac12\norm{x}^2$, then $\nabla f(v) = v$ $\Rightarrow$
$\ip{x-y}{\nabla f(y) - \nabla f(z)} = \ip{x-y}{y-z}$. Also
\begin{align*}
  2D_f(x,y) &= \norm{x}^2 - \norm{y}^2 - 2\ip{y}{x-y}
  = \ip{x}{x} - \ip{y}{x} - \ip{y}{x-y} \\
  &= \ip{x-y}{x} - \ip{y}{x-y} = \ip{x-y}{x-y} = \norm{x-y}^2
  \;\Rightarrow\; D_f(x,y) = \tfrac12\norm{x-y}^2 .
\end{align*}
As a result, for $f(x) = \frac12\norm{x}^2$, (a) becomes
\[
  \tfrac12\norm{x-z}^2 = \tfrac12\norm{x-y}^2 + \ip{x-y}{y-z} + \tfrac12\norm{y-z}^2 .
\]

\section*{Exercise 12}

$f : \R^d \to \R$, $f \in C^2(\R^d)$. Then $f$ is convex if and only if $\nabla^2 f(x) \succeq 0$.

\subsection*{1)}
Assume $\nabla^2 f(x) \succeq 0$ for all $x$. Then
\[
  \ip{\nabla f(x) - \nabla f(y)}{x-y}
  = \int_0^1 \ip{\nabla^2 f\bigl(y + t(x-y)\bigr)(x-y)}{x-y}\,dt .
\]
With $v := x - y$ we get $\int_0^1 \ip{\nabla^2 f(y + tv)\,v}{v}\,dt$, and because
$\nabla^2 f \succeq 0$, $\ip{\nabla^2 f(y + tv)\,v}{v} \ge 0$. As a result
$\ip{\nabla f(x) - \nabla f(y)}{x-y} \ge 0$, so $f$ is convex by Proposition 7
((iii) $\Rightarrow$ (i)).

\subsection*{2)}
If $f$ is convex, then $f(x) \ge f(y) + \ip{\nabla f(y)}{x-y}$. Define $x := y + hv$,
$h \in \R$, $v \in \R^d$. Then
\[
  f(y + hv) \ge f(y) + h\ip{\nabla f(y)}{v}.
\]
Also, because $f \in C^2(\R^d)$,
\[
  f(y + hv) = f(y) + h\ip{\nabla f(y)}{v} + \frac{h^2}{2}\ip{\nabla^2 f(y)\,v}{v} + o(h^2\norm{v}^2).
\]
As a result $\frac{h^2}{2}\ip{\nabla^2 f(y)\,v}{v} + o(h^2\norm{v}^2) \ge 0$ for all $h, v, y$.
Dividing by $\frac{h^2}{2}$ and letting $h \to 0$:
$\ip{\nabla^2 f(y)\,v}{v} \ge 0$ $\Rightarrow$ $\nabla^2 f(y) \succeq 0$ for all $y$.

\section*{Exercise 13}

$f : \R^d \to \R$ is convex, $x_i \in \R^d$, $\lambda_i \ge 0$ and $\sum_{i=1}^m \lambda_i = 1$.
Then $f\bigl(\sum_{i=1}^m \lambda_i x_i\bigr) \le \sum_{i=1}^m \lambda_i f(x_i)$.

\subsection*{1)}
For $m = 2$, $\lambda_1 := \lambda$, then $\lambda_2 = 1 - \lambda$, and
$f(\lambda x_1 + (1-\lambda)x_2) \le \lambda f(x_1) + (1-\lambda) f(x_2)$ by convexity.

\subsection*{2)}
If the inequality is true for $m = k$, we show it for $m = k+1$. If
$\lambda_k + \lambda_{k+1} = 0$, then $\lambda_k = \lambda_{k+1} = 0$ and this is the case
$m = k$. Otherwise let
\[
  \lambda'_k := \lambda_k + \lambda_{k+1}, \qquad
  x'_k := \frac{\lambda_k x_k + \lambda_{k+1} x_{k+1}}{\lambda_k + \lambda_{k+1}},
\]
and $\lambda'_i := \lambda_i$, $x'_i := x_i$ for $i = 1, \dots, k-1$. Then
\[
  f\Bigl(\sum_{i=1}^{k+1} \lambda_i x_i\Bigr) = f\Bigl(\sum_{i=1}^{k} \lambda'_i x'_i\Bigr)
  \le \sum_{i=1}^{k} \lambda'_i f(x'_i)
  = \sum_{i=1}^{k-1} \lambda_i f(x_i)
  + (\lambda_k + \lambda_{k+1})\, f\Bigl(\frac{\lambda_k x_k + \lambda_{k+1} x_{k+1}}{\lambda_k + \lambda_{k+1}}\Bigr). \tag{1}
\]
By the case $m = 2$,
\[
  f\Bigl(\frac{\lambda_k}{\lambda_k + \lambda_{k+1}} x_k + \frac{\lambda_{k+1}}{\lambda_k + \lambda_{k+1}} x_{k+1}\Bigr)
  \le \frac{\lambda_k}{\lambda_k + \lambda_{k+1}} f(x_k) + \frac{\lambda_{k+1}}{\lambda_k + \lambda_{k+1}} f(x_{k+1}). \tag{2}
\]
From (1) and (2),
\[
  f\Bigl(\sum_{i=1}^{k+1} \lambda_i x_i\Bigr)
  \le \sum_{i=1}^{k-1} \lambda_i f(x_i) + \lambda_k f(x_k) + \lambda_{k+1} f(x_{k+1})
  = \sum_{i=1}^{k+1} \lambda_i f(x_i).
\]

\section*{Exercise 14}

Let $f : \R^d \to \R$, $f \in C^1(\R^d)$. Then the following statements are equivalent:
\begin{itemize}
  \item[(i)] $f$ is $\mu$-convex
  \item[(ii)] $\mu\norm{x-y}^2 \le 2D_f(x,y)$ for all $x, y \in \R^d$
  \item[(iii)] $\mu\norm{x-y}^2 \le \ip{\nabla f(x) - \nabla f(y)}{x-y}$ for all $x, y \in \R^d$
\end{itemize}

\subsection*{(i) $\Rightarrow$ (ii)}
$g(x) := f(x) - \frac{\mu}{2}\norm{x}^2$, then $g$ is convex, as a result (Proposition 7)
\[
  D_g(x,y) = g(x) - g(y) - \ip{\nabla g(y)}{x-y} \ge 0,
\]
which is the same as
\[
  f(x) - f(y) - \ip{\nabla f(y)}{x-y}
  - \mu\Bigl(\frac{\norm{x}^2}{2} - \frac{\norm{y}^2}{2} - \ip{y}{x-y}\Bigr) \ge 0 .
\]
On the other hand $\frac{\norm{x}^2}{2} - \frac{\norm{y}^2}{2} - \ip{y}{x-y} = \frac12\norm{x-y}^2$.
And finally we get
\[
  f(x) - f(y) - \ip{\nabla f(y)}{x-y} \ge \frac{\mu}{2}\norm{x-y}^2 .
\]

\subsection*{(ii) $\Rightarrow$ (iii)}
\begin{align*}
  & f(x) - f(y) - \ip{\nabla f(y)}{x-y} \ge \tfrac{\mu}{2}\norm{x-y}^2 \\
+\; & f(y) - f(x) - \ip{\nabla f(x)}{y-x} \ge \tfrac{\mu}{2}\norm{x-y}^2 \\
=\; & -\ip{\nabla f(y)}{x-y} + \ip{\nabla f(x)}{x-y} \ge \mu\norm{x-y}^2,
\end{align*}
which is the same as $\ip{\nabla f(x) - \nabla f(y)}{x-y} \ge \mu\norm{x-y}^2$.

\subsection*{(iii) $\Rightarrow$ (i)}
First, if $\ip{\nabla g(x) - \nabla g(y)}{x-y} \ge 0$ for all $x, y$, then $g$ is convex:
\[
  g(y) = g(x) + \int_0^1 \ip{\nabla g\bigl(x + t(y-x)\bigr)}{y-x}\,dt .
\]
With $u := x + t(y-x)$, $t \in (0,1]$, we get
$\ip{\nabla g(u) - \nabla g(x)}{u-x} \ge 0$.
Expanding, $\ip{\nabla g(u)}{u-x} \ge \ip{\nabla g(x)}{u-x}$, and since $u - x = t(y-x)$,
dividing by $t$:
\[
  \ip{\nabla g(u)}{y-x} \ge \ip{\nabla g(x)}{y-x}.
\]
Integrating over $t$,
\[
  \int_0^1 \ip{\nabla g\bigl(x + t(y-x)\bigr)}{y-x}\,dt \ge \int_0^1 \ip{\nabla g(x)}{y-x}\,dt
  = \ip{\nabla g(x)}{y-x},
\]
as a result $g(y) - g(x) \ge \ip{\nabla g(x)}{y-x}$, so $g$ is convex by Proposition 7
((ii) $\Rightarrow$ (i)).

Now let $g(x) := f(x) - \frac{\mu}{2}\norm{x}^2$, so $\nabla g(x) = \nabla f(x) - \mu x$. Then
\[
  \ip{\nabla g(x) - \nabla g(y)}{x-y} = \ip{\nabla f(x) - \nabla f(y) - \mu(x-y)}{x-y} \ge 0
\]
is the same as $\ip{\nabla f(x) - \nabla f(y)}{x-y} \ge \mu\norm{x-y}^2$. So (iii) gives that
$g$ is convex, which is the same as $f$ being $\mu$-convex.

\section*{Exercise 16}

If $f : \R^d \to \R$, $f \in C^2(\R^d)$, then $f$ is $\mu$-convex iff
$\mu I \preceq \nabla^2 f(x)$ for all $x$.

$g(x) := f(x) - \frac{\mu}{2}\norm{x}^2$, then $g \in C^2(\R^d)$ and
$\nabla^2 g(x) = \nabla^2 f(x) - \mu I$.

\subsection*{1)}
If $\nabla^2 f(x) \succeq \mu I$, then $\nabla^2 g(x) \succeq 0$, then $g$ is convex
(Exercise 12), then $f$ is $\mu$-convex.

\subsection*{2)}
If $f$ is $\mu$-convex, then $g$ is convex, then $\nabla^2 g(x) \succeq 0$ (Exercise 12),
then $\nabla^2 f(x) \succeq \mu I$.
